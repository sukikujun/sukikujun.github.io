
## Oracle

### 結合

```sql
-- 部署テーブル
drop table if exists dept;
CREATE TABLE DEPT (
    DEPT_ID   NUMBER PRIMARY KEY,
    DEPT_NAME VARCHAR2(50)
);

-- 社員テーブル
drop table if exists emp;
CREATE TABLE EMP (
    EMP_ID    NUMBER PRIMARY KEY,
    EMP_NAME  VARCHAR2(50),
    DEPT_ID   NUMBER -- 部署ID（外部キー想定）
);

-- 部署データ
INSERT INTO DEPT VALUES (10, '営業部');
INSERT INTO DEPT VALUES (20, '開発部');

-- 社員データ（10番の営業部のみ配属）
INSERT INTO EMP VALUES (1, '田中', 10);
INSERT INTO EMP VALUES (2, '佐藤', 10);

COMMIT;
```

-- inner join
select a.*, b.emp_name  from dept a , emp b where a.DEPT_ID = b.DEPT_ID;

select a.*, b.emp_name  from dept a inner join emp b on a.DEPT_ID = b.DEPT_ID;

-- left outer
select a.*, b.emp_name  from dept a left OUTER join emp b on a.DEPT_ID = b.DEPT_ID;

select a.*, b.emp_name  from dept a , emp b where a.DEPT_ID = b.DEPT_ID(+);

-- cross join
select a.*, b.emp_name, b.emp_id, b.dept_id as ed_id  from dept a cross join emp b;

select a.*, b.emp_name, b.emp_id, b.dept_id as ed_id  from dept a, emp b;
