### SELECT

```plsql
declare
   date_time varchar2(50);
begin
   -- select 文の結果を変数に代入
   select sysdate
     into date_time
     from dual;
   dbms_output.put_line('現在の日付と時刻: ' || date_time);
end;
```

### IF 文

`else if` ではなく、`elsif` と書く

```psql
declare
   name    varchar2(50);
   message varchar2(100);
begin
   name := 'Yurushi';
   name := 'Inamori';
   if ( name = 'Yurushi' ) then
      message := 'こんにちは、Yurushiさん！';
   elsif ( name = 'Inamori' ) then
      message := 'こんにちは、Inamoriさん！';
   else
      message := 'こんにちは、ゲストさん！';
   end if;

   dbms_output.put_line(message);
end;
```

### LOOP

```plsql
declare
   type month_list is
      varray(12) of number;
   v_small_months month_list := month_list(
      2,
      4,
      6,
      9,
      11
   );
begin
   for i in 1..12 loop
      for j in 1..v_small_months.count loop
         if ( i = v_small_months(j) ) then
            dbms_output.put_line(v_small_months(j) || '月は「小の月」です。');
            exit;
         end if;
      end loop;
      dbms_output.put_line(i || '月は「大の月」です。');
   end loop;
end;
```

### EXCEPTION

```plsql
declare
   name varchar2(50);
   nonameexception exception;
begin
  --  name := '鈍色 聴';
   name := '';
   if name is not null then
      dbms_output.put_line('名前は'
                           || name || 'です。');
   else
      raise nonameexception;
   end if;
exception
   when nonameexception then
      dbms_output.put_line('名前が入力されていません。');
end;
```

### %TYPE 

変数のデータ型として%TYPEを用いると、テーブル側の設計変更でデータ型が変更されてもPL/SQL側の変数のデータ型を変更する必要がなくなるため、コードの保守性が向上します。

```plsql
declare
   depno   lixun.dept.dept_no%type;
   depname lixun.dept.dept_name%type;
begin
   select dept_no,
          dept_name
     into
      depno,
      depname
     from lixun.dept
    where dept_no = 1;
   dbms_output.put_line('dept_no: ' || depno);
   dbms_output.put_line('dept_name: ' || depname);
end;
```

### %ROWTYPE

```plsql
declare
   dept dept%rowtype;
begin
   select *
     into dept
     from dept
    where dept_no = 1;
   dbms_output.put_line('部署コード: ' || dept.dept_no);
   dbms_output.put_line('部署名: ' || dept.dept_name);
   dbms_output.put_line('都道府県: ' || dept.prefecture);
end;
```
